ssh-keygen: create and inspect SSH keys. A key pair = a private key (stays on your machine) + a .pub (you share it).

ssh-keygen -t ed25519 -C 'you@example.com'    # new key pair (ed25519 is modern, short and fast); asks where to save and for a passphrase
ssh-keygen -t ed25519 -f ~/.ssh/work -C work  # choose the file name (-f): creates ~/.ssh/work and ~/.ssh/work.pub
ssh-keygen -t rsa -b 4096                     # RSA, only for old servers that do not accept ed25519
ssh-keygen -p -f ~/.ssh/id_ed25519            # change or add a passphrase on an existing key (-p)
ssh-keygen -l -f ~/.ssh/id_ed25519.pub        # show the fingerprint (-l): compare it with what GitHub or the server shows
ssh-keygen -y -f ~/.ssh/id_ed25519 > id.pub   # recreate the public key from a private key (-y)
ssh-keygen -R host                            # forget a host in known_hosts (-R): after a server was rebuilt and ssh warns that its identification changed
ssh-keygen -F host                            # look a host up in known_hosts (-F)
pbcopy < ~/.ssh/id_ed25519.pub                # copy the PUBLIC key to the clipboard, to paste into GitHub (Settings, SSH keys)
chmod 600 ~/.ssh/id_ed25519                   # private keys must not be readable by others, or ssh refuses to use them
ssh-keygen -N '' -t ed25519 -f ./k            # no passphrase (-N ''): fine for a throwaway test key, not for your real one
