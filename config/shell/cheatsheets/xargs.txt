GNU xargs: turn lines of input into arguments for another command.

fd -e md | xargs -r wc -l                        # run wc -l once with all the Markdown files as arguments
echo a b c | xargs -n 1 echo                  # one argument per run (-n 1)
fd -e png | xargs -r -I{} cp {} /tmp/            # put each item where {} is (-I{}): runs once per item
fd -e md | xargs -r -P 4 -n 1 prettier --check   # run up to 4 at a time in parallel (-P)
fd -0 -e md | xargs -0 -r ls -l                  # NUL-separated (-0) is safe for names with spaces; pair with fd -0 or find -print0
fd -e tmp | xargs -r -p echo                     # ask before each run (-p)
git branch --merged | xargs -r git branch -d     # a real example: delete merged branches (read the list first)
lsof -ti :3000 | xargs -r kill                   # free a port: kill the process on it
GNU xargs runs once on empty input by default: use -r to skip that run.
Use find -print0 / fd -0 with xargs -0 for file names containing whitespace or newlines.
xargs --version                              # confirm GNU Findutils
